# LYVRA Suno / Studio 2 — Dual Control Hierarchy

STATUS: CURRENT_PRODUCTIVE_NATIVE_RELATION
TRACK_DESIGN_VERSION: v3.4
TRACK_LOGICAL_REVISION: 90

## Core finding

There are TWO distinct creator-facing control families.

NORMAL_TRACK_CONTROL_SURFACE != STUDIO2_OPERATION_CONTROL_SURFACE

They share some visible parameters but do not share the same role, decision hierarchy, or scope.

---

## CONTROL FAMILY A — NORMAL TRACK / GENERATION CONTROL SURFACE

ROLE = choose renderer/model and generation behavior for a track-level render request.
LOCATION = creator-facing surface separate from content boxes.
CONTENT_BOXES_NE_CONTROL_SURFACE = true

### Hierarchy A0 — musical intent and current track state
INPUTS = TRACK_INTENT | GENRE_DNA | MOTOR_DNA | VOICE_INTENT | STRUCTURAL_COMPLEXITY | IDENTITY_REQUIREMENT | CONTROL_NEED | EXPLORATION_NEED | CURRENT_RENDERER_EVIDENCE

### Hierarchy A1 — renderer/model decision
MODEL_DECISION = CONTEXTUAL_PER_TRACK
NATIVE_MODELS = v6 | v6-wild | v6-mini
CUSTOM_MODELS = LYVRA_STYLE | 666PoWeRSoUnD | 666SD_PsyTrance
CUSTOM_MODEL_NE_NATIVE_MODEL_EVIDENCE = true
PRODUCTION_SUITABILITY_NE_LEARNING_VALUE = true
MODEL_COVERAGE_REQUIRED = true

### Hierarchy A2 — major mode / option decisions
OPTIONS = MAX_MODE | INSTRUMENTAL | PERSONALIZATION | VOICE_GENDER_IF_RELEVANT | EXCLUDED_STYLES_IF_RELEVANT
MAX_MODE_DECISION_SEPARATE_FROM_MODEL_IDENTITY = true
PERSONALIZATION_DECISION_SEPARATE_FROM_MODEL_IDENTITY = true

### Hierarchy A3 — creativity / adherence controls
CONTROLS = VARIETY | WEIRDNESS | STYLE_INFLUENCE
VALUES = CONTEXTUAL_NOT_FIXED_PRESET
OLD_RED_ZONE_THRESHOLDS_NE_CURRENT_FACT_WITHOUT_REVALIDATION = true

### Hierarchy A4 — conditioning controls
AUDIO_INFLUENCE_IF_APPLICABLE = CONTEXTUAL
SOURCE_AUDIO_RELATION_MUST_BE_EXPLICIT = true

### Hierarchy A5 — duration
TARGET_DURATION = CONTEXTUAL
DURATION_NE_FIXED_UI_ASSUMPTION = true
CURRENT_UI_CONFIRMATION_REQUIRED_WHEN_MATERIAL = true

### Hierarchy A6 — renderer preaudit and plausibility
PREAUDIT = DRIFT_RISK + RENDERER_INTERPRETABILITY
FRESH_WEB_CHECK_WHEN_MATERIAL = true
OFFICIAL_CURRENT_FIRST = true
CURRENT_COMMUNITY_SECOND = true
OWN_RENDER_EVIDENCE = HIGHEST_PRACTICAL_OBSERVATION
PLAUSIBLE_RENDERER_TRANSLATION_GATE = REQUIRED

## CONTROL FAMILY B — STUDIO 2 OPERATION CONTROL SURFACE

ROLE = control a concrete Studio 2 edit/production operation inside an existing project.
STUDIO_CONTROL_SURFACE_NE_NORMAL_TRACK_SLIDER_SURFACE = true
VALUES = TASK_DEPENDENT

### Hierarchy B0 — task class
TASK_CLASS = PRESERVE | REPAIR | CREATIVE_MUTATION | NEW_LAYER | TRANSITION | STEM_REBUILD | MIX_OPERATION | AUTOMATION_OPERATION | FX_OR_PLUGIN_OPERATION

### Hierarchy B1 — preserve / mutate decision
PRESERVE_MAP = WHAT_MUST_SURVIVE
MUTATION_MAP = WHAT_MAY_CHANGE
PRESERVE_EXCELLENT_MATERIAL_AND_REPAIR_TARGETED_FAILURE = true

### Hierarchy B2 — Studio surface selection
VISIBLE_SURFACES = STYLE | SONGTEXT | STEUERUNG | STUDIO_CHAT | TIMELINE | FX | CUSTOM_PLUGINS
SELECT_SURFACE_BY_TASK = true

### Hierarchy B3 — edit scope
EDIT_SCOPE = WHOLE_TRACK | TIMELINE_REGION | STEM | AUDIO_TRACK | MIDI_TRACK | AUTOMATION | EFFECT_CHAIN | CUSTOM_PLUGIN | TRANSITION
WHOLE_SECTION_REGENERATION_FOR_LOCAL_FAILURE = DISFAVORED

### Hierarchy B4 — operation-specific controls
OBSERVED_STUDIO_CONTROLS = WEIRDNESS | STYLE_INFLUENCE | AUDIO_INFLUENCE
THESE_CONTROLS_NE_NORMAL_TRACK_GLOBAL_PRESET = true
CONTROL_MEANING_DEPENDS_ON_TASK_AND_SCOPE = true

### Hierarchy B5 — specialized sub-control trees
STEM_REPAIR = IDENTIFY > ISOLATE > PRESERVE_GOOD > REPAIR_BAD > READBACK
TRANSITION = FROM_STATE > TRANSFORMATION > MEMORY > DESTABILIZATION > HANDOFF > TO_STATE
MIX = INSPECT > STEM_SPLIT > DEFINE_STEM_ROLE > PROTECT_MOTOR > FX_CHAINS > SIDECHAIN > EQ_SPECTRAL_SPACE > SPACE_DELAY_DISTORTION > AUTOMATION > LISTENING_CHECK > MIX_CANDIDATE
CUSTOM_EFFECT_PLUGIN = EFFECT_TARGET > MUSICAL_FUNCTION > AUDIO_ROLE > LOW_MID_HIGH_PROTECTION > PSYCHOACOUSTIC_TARGET > DSP_ARCHITECTURE > PROCESSING_ORDER > CONTROL_SURFACE > PARAMETER_RANGES > SAFETY > BUILDER_PROMPT > PLAN_AUDIT > BUILD > CONTROL_TEST > LISTEN > REVISE > LEARN

### Hierarchy B6 — execution loop
DAEMON_FLOW = CURRENT_PROJECT_STATE > GOAL > PRESERVE_MUTATE > SELECT_SURFACE > GENERATE_EXACT_INPUT > USER_EXECUTES > LISTEN_OBSERVE > PASS_PARTIAL_FAIL > TARGETED_NEXT_STEP
DAEMON_DECISION_AUTHORITY = NONE

### Hierarchy B7 — evidence / validation
STUDIO2_OPERATION_RESULT_NE_DURABLE_TRUTH = true
RESULT_CLASS = PASS | PARTIAL | FAIL | UNVERIFIED
STUDIO2_TO_ANALYTICS = CLASSIFIED_OPERATION_EVIDENCE
ANALYTICS_TO_MUSIC_MEMORY = REVIEWED_CAUSAL_LEARNING_WITH_SCOPE
